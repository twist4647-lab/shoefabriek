var KEYS = ['id','ts','inspector','product','style','date','section','process','operator','machine','pairs','serial_start','serial_end','dl','dr','serial_defects','comments'];
function sheet_() { return SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Entries'); }

function doGet(e) {
  var rows = sheet_().getDataRange().getValues(), tz = Session.getScriptTimeZone(), out = [];
  for (var i = 1; i < rows.length; i++) {
    var o = {};
    KEYS.forEach(function(k, j) {
      var v = rows[i][j];
      if (v instanceof Date) v = Utilities.formatDate(v, tz, 'yyyy-MM-dd');
      o[k] = v;
    });
    if (o.id) out.push(o);
  }
  return ContentService.createTextOutput(JSON.stringify({entries: out}))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(20000);
  var s = sheet_(), d = JSON.parse(e.postData.contents);
  var rows = s.getDataRange().getValues();
  var id = d.action === 'delete' ? d.id : d.entry.id, row = 0;
  for (var i = 1; i < rows.length; i++) {
    if (String(rows[i][0]) === String(id)) { row = i + 1; break; }
  }
  if (d.action === 'delete') {
    if (row) s.deleteRow(row);
  } else {
    var vals = KEYS.map(function(k) { return d.entry[k] === undefined ? '' : d.entry[k]; });
    if (row) s.getRange(row, 1, 1, KEYS.length).setValues([vals]);
    else s.appendRow(vals);
  }
  lock.releaseLock();
  return ContentService.createTextOutput('ok');
}
